cmake_minimum_required(VERSION 3.20)
project(satp64 CXX)

cmake_policy(SET CMP0135 NEW)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

include(FetchContent)
FetchContent_Declare(
    googletest
    URL https://github.com/google/googletest/archive/refs/tags/v1.14.0.zip
)
set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(googletest)

include_directories(include)

add_library(satp64_lib src/satp64.cpp)

add_executable(satp64_runner src/main.cpp)
target_link_libraries(satp64_runner PRIVATE satp64_lib)

enable_testing()
add_executable(satp64_tests tests/satp64_tests.cpp)
target_link_libraries(satp64_tests PRIVATE satp64_lib gtest_main)

include(GoogleTest)
gtest_discover_tests(satp64_tests)
